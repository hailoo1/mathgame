       function updateBuilderQuestion() {

           const slotFood = document.getElementById('slot-food');

           const slotNumber = document.getElementById('slot-number');

           const slotRatio = document.getElementById('slot-ratio');

           const questionEl = document.getElementById('builder-question');

           const interactiveEl = document.getElementById('builder-interactive');

           const inputLabel = document.getElementById('builder-input-label');

           const feedbackEl = document.getElementById('builder-feedback');

           const expEl = document.getElementById('builder-explanation');

           if (slotFood) slotFood.innerText = state.selectedFood;

           if (slotNumber) slotNumber.innerText = `${state.selectedNumber}개`;

           if (slotRatio) slotRatio.innerText = state.selectedRatio;

           const [a, b] = state.selectedRatio.split(':').map(Number);

           const sumParts = a + b;

           const total = state.selectedNumber;

           // Generate clean problem

           const isDivisible = (total % sumParts === 0);

           const questionText = `선생님이 ${state.selectedFood} ${total}개를 두 모둠에게 ${a} : ${b} 비율로 나누어 주려고 합니다. 첫 번째 모둠이 받는 ${state.selectedFood}는 모두 몇 개일까요?`;

           if (questionEl) questionEl.innerText = questionText;

           if (interactiveEl) interactiveEl.classList.remove('hidden');

           if (feedbackEl) feedbackEl.innerText = '';

         

           if (inputLabel) {

               inputLabel.innerText = isDivisible ?

                   `정답을 구해보세요! (${total}개를 ${a}:${b}로 분배)` :

                   `💡 참고: ${total}개는 ${sumParts}의 배수가 아니어서 소수/분수 답이 나올 수 있습니다.`;

           }

           const answerVal = (total * a) / sumParts;

           if (expEl) {

               expEl.classList.remove('hidden');

               expEl.innerHTML = `

                   <div class="font-bold text-indigo-300 mb-1"><i class="fa-solid fa-lightbulb mr-1"></i>풀이 힌트 & 정답</div>

                   • 비의 총합: ${a} + ${b} = ${sumParts}<br>

                   • 첫 번째 모둠의 몫 = ${total} × (${a} / ${sumParts}) = <span class="font-bold text-amber-300">${Number.isInteger(answerVal) ? answerVal : answerVal.toFixed(1)}개</span>

               `;

           }

       }

       window.checkBuilderAnswer = function() {

           const input = document.getElementById('builder-answer-input');

           const feedbackEl = document.getElementById('builder-feedback');

           if (!input || !feedbackEl) return;

           const userAns = parseFloat(input.value);

           const [a, b] = state.selectedRatio.split(':').map(Number);

           const expected = (state.selectedNumber * a) / (a + b);

           if (isNaN(userAns)) {

               feedbackEl.innerHTML = '<span class="text-amber-400">숫자를 입력해주세요!</span>';

               return;

           }

           if (Math.abs(userAns - expected) < 0.01) {

               feedbackEl.innerHTML = '<span class="text-emerald-400"><i class="fa-solid fa-circle-check mr-1"></i>정답입니다! 참 잘했어요! 🎉</span>';

               addStars(10);

           } else {

               feedbackEl.innerHTML = `<span class="text-rose-400"><i class="fa-solid fa-circle-xmark mr-1"></i>아쉬워요! 정답은 ${Number.isInteger(expected) ? expected : expected.toFixed(1)}개 입니다.</span>`;

           }

       };

       function generateNextQuiz() {

           const food = state.foods[Math.floor(Math.random() * state.foods.length)];

           const ratioStr = state.ratios[Math.floor(Math.random() * state.ratios.length)];

           const [a, b] = ratioStr.split(':').map(Number);

           const sumParts = a + b;

           // Guarantee an integer answer by picking a multiple of (a + b)

           const multiplier = Math.floor(Math.random() * 6) + 1; // 1 to 6

           const total = sumParts * multiplier;

           const correctAnswer = a * multiplier;

           // Generate options

           const options = new Set([correctAnswer]);

           while (options.size < 4) {

               const wrong = Math.max(1, correctAnswer + (Math.floor(Math.random() * 7) - 3));

               if (wrong !== correctAnswer) options.add(wrong);

           }

           state.currentQuiz = {

               food, total, a, b, ratioStr,

               correctAnswer,

               options: Array.from(options).sort(() => Math.random() - 0.5)

           };

           renderQuizUI();

       }

       function initCanvasSystem() {

           penCanvas = document.getElementById('pen-canvas');

       function resizePenCanvas() {

           if (!penCanvas) return;

           const wrapper = document.getElementById('canvas-wrapper');

           if (!wrapper) return;

           const rect = wrapper.getBoundingClientRect();

           if (rect.width <= 0 || rect.height <= 0) return;

           // Preserve canvas drawing on resize

           const tempCanvas = document.createElement('canvas');

           tempCanvas.width = penCanvas.width;

           tempCanvas.height = penCanvas.height;

           const tempCtx = tempCanvas.getContext('2d');

           if (penCanvas.width > 0 && penCanvas.height > 0) {

               tempCtx.drawImage(penCanvas, 0, 0);

           }

           penCanvas.width = rect.width;

           penCanvas.height = rect.height;

           if (tempCanvas.width > 0 && tempCanvas.height > 0) {

               penCtx.drawImage(tempCanvas, 0, 0);

           }

       }

       function getCanvasCoords(e) {

           const rect = penCanvas.getBoundingClientRect();

           return {

               x: e.clientX - rect.left,

               y: e.clientY - rect.top

           };

       }

       function startDrawing(e) {

           if (state.canvasMode !== 'pen' && state.canvasMode !== 'eraser') return;

           isDrawing = true;

           const coords = getCanvasCoords(e);

           lastX = coords.x;

           lastY = coords.y;

       }

       function draw(e) {

           if (!isDrawing || (state.canvasMode !== 'pen' && state.canvasMode !== 'eraser')) return;

           const coords = getCanvasCoords(e);

           penCtx.beginPath();

           penCtx.moveTo(lastX, lastY);

           penCtx.lineTo(coords.x, coords.y);

           if (state.canvasMode === 'eraser') {

               penCtx.globalCompositeOperation = 'destination-out';

               penCtx.lineWidth = state.penSize * 4;

           } else {

               penCtx.globalCompositeOperation = 'source-over';

               penCtx.strokeStyle = state.penColor;

               penCtx.lineWidth = state.penSize;

               penCtx.lineCap = 'round';

               penCtx.lineJoin = 'round';

           }

           penCtx.stroke();

           lastX = coords.x;

           lastY = coords.y;

       }

       function stopDrawing() {

           isDrawing = false;

       }

       function makeElementDraggable(el) {

           let active = false;

           let currentX, currentY, initialX, initialY;

           let xOffset = 0, yOffset = 0;

           el.addEventListener('pointerdown', dragStart);

           function dragStart(e) {

               if (state.canvasMode === 'pen' || state.canvasMode === 'eraser') return;

               if (e.target.tagName === 'BUTTON' || e.target.closest('button')) return;

               initialX = e.clientX - xOffset;

               initialY = e.clientY - yOffset;

               active = true;

               document.addEventListener('pointermove', drag);

               document.addEventListener('pointerup', dragEnd);

           }

           function drag(e) {

               if (!active) return;

               e.preventDefault();

               currentX = e.clientX - initialX;

               currentY = e.clientY - initialY;

               xOffset = currentX;

               yOffset = currentY;

               el.style.transform = `translate3d(${currentX}px, ${currentY}px, 0)`;

           }

           function dragEnd() {

               initialX = currentX;

               initialY = currentY;

               active = false;

               document.removeEventListener('pointermove', drag);

               document.removeEventListener('pointerup', dragEnd);

           }

       }

       window.exportCanvasToPNG = function() {

           if (!penCanvas) return;

           // Render composite export canvas

           const expCanvas = document.createElement('canvas');

           expCanvas.width = penCanvas.width;

           expCanvas.height = penCanvas.height;

           const expCtx = expCanvas.getContext('2d');

           // Draw Background

           expCtx.fillStyle = '#ffffff';

           expCtx.fillRect(0, 0, expCanvas.width, expCanvas.height);

           // Draw Pen Strokes

           expCtx.drawImage(penCanvas, 0, 0);

           // Render Card / Text Elements

           const layer = document.getElementById('canvas-elements-layer');

           if (layer) {

               const elements = layer.children;

               for (let el of elements) {

                   const rect = el.getBoundingClientRect();

                   const layerRect = layer.getBoundingClientRect();

                   const x = rect.left - layerRect.left;

                   const y = rect.top - layerRect.top;

                   expCtx.save();

                   expCtx.fillStyle = '#f8fafc';

                   expCtx.strokeStyle = '#1e293b';

                   expCtx.lineWidth = 3;

                 

                   // Simple Box Representation

                   expCtx.beginPath();

                   expCtx.roundRect(x, y, rect.width, rect.height, 12);

                   expCtx.fill();

                   expCtx.stroke();

                   // Text

                   expCtx.fillStyle = '#0f172a';

                   expCtx.font = '20px Jua, sans-serif';

                   const textStr = el.innerText.replace('x', '').trim();

                   expCtx.fillText(textStr, x + 12, y + rect.height / 2 + 6);

                   expCtx.restore();

               }

           }

           // Download Trigger

           const link = document.getElementById('download-link') || document.createElement('a');

           link.download = `math-card-problem-${Date.now()}.png`;

           link.href = expCanvas.toDataURL('image/png');

           link.click();

       };

       function setupFirebaseListeners() {

           const { db, doc, onSnapshot } = window.firebaseSync;

           if (!db || !state.roomCode) return;

           // Room listener setup

           const roomRef = doc(db, 'rooms', state.roomCode);

           onSnapshot(roomRef, (snapshot) => {

               if (snapshot.exists()) {

                   const data = snapshot.data();

                   if (state.syncRole === 'student' && data.numbers) {

                       state.numbers = data.numbers;

                       renderCardBoard();

                   }

               }

           });

       }
